🎯 WHAT IT PREDICTS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
QUESTION: Will market price go UP, DOWN, or stay NEUTRAL after this news?

INPUT:   News article about a Polymarket event
OUTPUT:  Prediction + Confidence level

EXAMPLE:
  Article: "Bitcoin surges to $95K"
  → Prediction: UP ↗ (78% confidence)
  → Action: BUY (confidence > 65%)

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
🔍 8 INPUT FEATURES (What it analyzes)

2. sentiment_magnitude   (0 to 10+)  Strength of emotion
3. source_credibility    (0 to 1)    Bloomberg=1.0, blog=0.5
4. title_length          (chars)     Longer = more detail
5. has_description       (0 or 1)    Article has body text?
6. keyword_overlap       (0 to 10)   How many words match market?
7. market_volume         ($)         Trading activity
8. market_volume_log     (log scale) Normalized volume

🔧 HOW TO USE IT
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
# Load the model
import pickle
with open('real_data_model.pkl', 'rb') as f:
    data = pickle.load(f)
    model = data['model']
    scaler = data['scaler']

# Prepare features (8 values)
features = [sentiment, magnitude, credibility, length,
            description, overlap, volume, volume_log]

# Predict
features_scaled = scaler.transform([features])
prediction = model.predict(features_scaled)[0]  # 1, 0, or -1
confidence = model.predict_proba(features_scaled).max()

# Decision
if confidence > 0.65:
    if prediction == 1:
        action = "BUY"
    elif prediction == -1:
        action = "SELL"
else:
    action = "HOLD"
